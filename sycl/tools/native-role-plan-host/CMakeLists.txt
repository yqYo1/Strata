cmake_minimum_required(VERSION 3.18)
project(strata_native_role_plan_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(STRATA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_library(native_role_plan STATIC "${STRATA_ROOT}/src/artifact/native_role_plan.cpp")
target_include_directories(native_role_plan PUBLIC "${STRATA_ROOT}/include")
target_compile_options(native_role_plan PRIVATE -Wall -Wextra -Wpedantic)
add_executable(native_role_plan_fixture native_role_plan_fixture.cpp)
target_link_libraries(native_role_plan_fixture PRIVATE native_role_plan)
target_compile_options(native_role_plan_fixture PRIVATE -Wall -Wextra -Wpedantic)
