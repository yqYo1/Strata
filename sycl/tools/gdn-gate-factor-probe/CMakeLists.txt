cmake_minimum_required(VERSION 3.24)
project(gdn_gate_factor_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
 message(FATAL_ERROR "Pinned IntelLLVM SYCL compiler required")
endif()
if(CMAKE_CXX_FLAGS MATCHES "fast-math|relaxed-math|fp-model[= ]fast")
 message(FATAL_ERROR "Fast/relaxed math forbidden")
endif()
add_executable(gdn_gate_factor_probe gdn_gate_factor_probe.cpp)
target_compile_features(gdn_gate_factor_probe PRIVATE cxx_std_20)
target_compile_options(gdn_gate_factor_probe PRIVATE -fsycl -fp-model=precise -fsycl-default-sub-group-size=32 -fsycl-device-code-split=per_kernel)
target_link_options(gdn_gate_factor_probe PRIVATE -fsycl -fsycl-device-code-split=per_kernel
 -Xsycl-target-backend=spir64 -cl-fp32-correctly-rounded-divide-sqrt)
