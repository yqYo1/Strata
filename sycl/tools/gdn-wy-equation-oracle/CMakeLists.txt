cmake_minimum_required(VERSION 3.20)
project(gdn_wy_equation_oracle LANGUAGES CXX)
add_executable(gdn_wy_equation_oracle gdn_wy_equation_oracle.cpp)
target_compile_features(gdn_wy_equation_oracle PRIVATE cxx_std_20)
set_target_properties(gdn_wy_equation_oracle PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang|IntelLLVM")
  target_compile_options(gdn_wy_equation_oracle PRIVATE -fno-fast-math -ffp-contract=off)
elseif(MSVC)
  target_compile_options(gdn_wy_equation_oracle PRIVATE /fp:strict)
else()
  message(FATAL_ERROR "Review strict floating-point compiler flags before admitting another compiler")
endif()
if(CMAKE_CXX_FLAGS MATCHES "ffast-math|Ofast|fsycl|fp-model[= ]fast")
  message(FATAL_ERROR "No fast math or SYCL flags admitted")
endif()
# Standalone portable host arithmetic: no parent project, production TU, model, ggml or GPU library.
