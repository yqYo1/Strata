cmake_minimum_required(VERSION 3.24)
project(iq2s_nt1_index_spread LANGUAGES C CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM" OR NOT CMAKE_C_COMPILER_ID STREQUAL "IntelLLVM")
  message(FATAL_ERROR "Requires reviewed icx/icpx IntelLLVM recipe")
endif()
foreach(flags CMAKE_C_FLAGS CMAKE_CXX_FLAGS CMAKE_EXE_LINKER_FLAGS)
  if("${${flags}}" MATCHES "fast-math|fsycl|march|ffp-contract=fast")
    message(FATAL_ERROR "Unreviewed global FP/ISA/SYCL flags")
  endif()
endforeach()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GGML_SOURCE_DIR "" CACHE PATH "Explicit pinned ggml source directory (ending ggml)")
if(NOT IS_DIRECTORY "${GGML_SOURCE_DIR}/src")
  message(FATAL_ERROR "Caller must supply GGML_SOURCE_DIR")
endif()
# Source hash checks, no Git invocation or dependency download.
foreach(item "src/ggml-cpu/arch/x86/quants.c|3c489fdc77e3ab484a5f188bff9c60c0e9483c951ed65c1a562a04c249d5628e" "src/ggml-common.h|0061131b615c5721fc88a78feeb22c1f8c450f1c2646a317d80796a653bf595c" "src/ggml-cpu/ggml-cpu.c|7b451e6528cec3bc27bfd812a1314859c92d22a00c2db371d543cde5b7d5ca94" "src/ggml-quants.c|07143d7068936ae46b3c528b2f3d4bbb666e74d88992165716174d243573965d")
  string(REPLACE "|" ";" pair "${item}")
  list(GET pair 0 path)
  list(GET pair 1 expected)
  file(SHA256 "${GGML_SOURCE_DIR}/${path}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Pinned source mismatch: ${path}")
  endif()
endforeach()
foreach(option CUDA SYCL HIP VULKAN METAL MUSA CANN WEBGPU ZDNN VIRTGPU VIRTGPU_BACKEND RPC OPENVINO ET ET_SYSEMU OPENCL HEXAGON ZENDNN BLAS ACCELERATE OPENMP OPENMP_FETCH BACKEND_DL CPU_ALL_VARIANTS BUILD_TESTS BUILD_EXAMPLES BUILD_SERVER)
  set(GGML_${option} OFF CACHE BOOL "CPU-only fixture" FORCE)
endforeach()
set(GGML_NATIVE ON CACHE BOOL "Production native CPU dependency" FORCE)
set(GGML_CPU ON CACHE BOOL "CPU backend" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static closure" FORCE)
add_compile_options(-fp-model=precise)
add_subdirectory("${GGML_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/ggml")
find_package(Threads REQUIRED)
add_executable(iq2s_index_spread iq2s_index_spread.cpp)
target_include_directories(iq2s_index_spread PRIVATE "${GGML_SOURCE_DIR}/include" "${GGML_SOURCE_DIR}/src" "${GGML_SOURCE_DIR}/src/ggml-cpu")
target_compile_options(iq2s_index_spread PRIVATE -mavx2 -mfma -mf16c)
target_link_libraries(iq2s_index_spread PRIVATE ggml-cpu ggml-base Threads::Threads)
