cmake_minimum_required(VERSION 3.24)
project(strata_native_service_host LANGUAGES C CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM" OR NOT CMAKE_C_COMPILER_ID STREQUAL "IntelLLVM")
  message(FATAL_ERROR "Reviewed host recipe requires IntelLLVM/icpx; no compiler substitution")
endif()
if(CMAKE_CXX_FLAGS MATCHES "fsycl|march" OR CMAKE_EXE_LINKER_FLAGS MATCHES "fsycl")
  message(FATAL_ERROR "No global SYCL or march flags are admitted on Strata host TUs")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(STRATA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." CACHE PATH "Frozen role-plan worktree")
set(GGML_SOURCE_DIR "/home/yayoi/orca/workspaces/Strata/ggml-sycl-pinned/ggml" CACHE PATH "Pinned ggml sources")
execute_process(COMMAND git -C "${GGML_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE GGML_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE GIT_RESULT)
if(NOT GIT_RESULT EQUAL 0 OR NOT GGML_HEAD STREQUAL "3cf03257f219afbe7334045ff7c6a06ac68c627d")
  message(FATAL_ERROR "ggml commit pin mismatch")
endif()
foreach(option CUDA SYCL HIP VULKAN METAL MUSA CANN WEBGPU ZDNN VIRTGPU VIRTGPU_BACKEND RPC OPENVINO ET ET_SYSEMU OPENCL HEXAGON ZENDNN BLAS ACCELERATE OPENMP OPENMP_FETCH BACKEND_DL CPU_ALL_VARIANTS BUILD_TESTS BUILD_EXAMPLES BUILD_SERVER)
  set(GGML_${option} OFF CACHE BOOL "CPU-only harness" FORCE)
endforeach()
set(GGML_NATIVE ON CACHE BOOL "Pinned native ggml CPU recipe" FORCE)
set(GGML_CPU ON CACHE BOOL "CPU backend" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static closure" FORCE)
add_subdirectory("${GGML_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/ggml")
find_package(Threads REQUIRED)
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_FLAGS "-mavx2 -mfma -mf16c")
check_cxx_source_compiles("#include <immintrin.h>
__attribute__((target(\"avxvnni\"))) static __m256i dots(__m256i a, __m256i b) { return _mm256_dpbusd_avx_epi32(_mm256_dpwssd_avx_epi32(a,a,b),a,b); }
int main() { volatile int x=1; return _mm256_extract_epi32(dots(_mm256_set1_epi32(x),_mm256_set1_epi32(x)),0)==0; }" Q2_AVXVNNI)
unset(CMAKE_REQUIRED_FLAGS)
set(CPU "${STRATA_ROOT}/src/kernels/cpu")
set(IQ_WRAPPER "${STRATA_ROOT}/sycl/src/kernels/cpu/iq_avx2.cpp")
add_executable(native_service_host native_service_host.cpp
  "${STRATA_ROOT}/src/artifact/native_role_plan.cpp"
  "${CPU}/expert_layout.cpp" "${CPU}/native_expert.cpp" "${CPU}/pool.cpp"
  "${CPU}/expert.cpp" "${CPU}/q2_avx2.cpp" "${IQ_WRAPPER}"
  "${CPU}/iq_avx512.cpp" "${CPU}/kq_avx2.cpp" "${CPU}/kq_avx1.cpp")
target_include_directories(native_service_host PRIVATE "${STRATA_ROOT}/include" "${GGML_SOURCE_DIR}/include" "${GGML_SOURCE_DIR}/src" "${GGML_SOURCE_DIR}/src/ggml-cpu")
target_compile_options(native_service_host PRIVATE -fp-model=precise)
target_compile_definitions(native_service_host PRIVATE STRATA_NATIVE_EXPERTS=1 HOST_Q2_AVXVNNI=$<BOOL:${Q2_AVXVNNI}>)
set_source_files_properties("${CPU}/expert.cpp" "${CPU}/iq_avx512.cpp" PROPERTIES COMPILE_OPTIONS "-mavx512f;-mavx512bw;-mavx512vl;-mavx512dq;-mavx512vnni;-mavx512vbmi;-mf16c;-mfma")
set_source_files_properties("${CPU}/q2_avx2.cpp" "${IQ_WRAPPER}" "${CPU}/kq_avx2.cpp" PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-mf16c")
set_source_files_properties("${CPU}/kq_avx1.cpp" PROPERTIES COMPILE_OPTIONS "-mavx")
set_source_files_properties("${CPU}/q2_avx2.cpp" PROPERTIES COMPILE_DEFINITIONS "STRATA_AVXVNNI=$<BOOL:${Q2_AVXVNNI}>")
set_source_files_properties("${IQ_WRAPPER}" PROPERTIES COMPILE_DEFINITIONS "STRATA_AVXVNNI=0")
# STRATA_IQ2S_GCC must remain undefined (the wrapper tests defined(), not its value).
target_link_libraries(native_service_host PRIVATE ggml-cpu ggml-base Threads::Threads)
