def worker_info(startup, engine, extra, workers):
    require(startup[-1].split() == ['READY', '262144', 'stop'], 'exact physical-context READY')
    lines = [s for s in startup if s.startswith('INFO ')]
    require(len(lines) == 1, 'one actual INFO')
    result = {}
    for word in lines[0].split()[1:]:
        require('=' in word, 'INFO key/value field')
        key, value = word.split('=', 1)
        require(key not in result, 'duplicate INFO field')
        result[key] = value
    slots = result.get('expert_slots', '')
    require(re.fullmatch(r'[0-9]+', slots) and 0 < int(slots) <= 48 * 512,
            'actual positive integer expert slots within model population')
    require(workers in (3,4,5), 'bounded worker sweep')
    extra = dict(extra)
    require(extra.get('pool_workers', '5') == '5', 'admitted baseline extra workers5')
    extra['pool_workers'] = str(workers)
    expected = dict(context='262144', kv='int8', kv_resident='32768', spec='4' if engine == 'xe' else '8',
                    lookup='0', pool_workers=str(workers), pcie_frac='0.00', conversation_cache_mib='0')
    require(all(k not in expected or str(v) == expected[k] for k, v in extra.items()), 'Main INFO cannot override fixed policy')
    expected.update(extra)
    require(all(result.get(k) == str(v) for k, v in expected.items()), 'actual effective fixed settings')
    require(int(result['vram_free_mib']) >= 512, 'READY free VRAM >=512MiB')
    require(result.get('batch_slots', '0') == '0', 'ordinary serial request only')
    return result

                        observed = monitor.check(stderr)
                        if observed['severe_messages']:
                            r['active_runtime_failure_diagnostics'] = observed
                            save()
                            raise ValueError('active post-READY runtime error')
                        next_diagnostic = time.monotonic() + 1
                    if time.monotonic() >= next_progress:
                        elapsed = time.monotonic() - entry['spawn_begin_monotonic']
                        size = stderr.stat().st_size
                        r.setdefault('owned_progress', []).append(dict(elapsed_seconds=elapsed, stderr_bytes=size))
                        require(len(r['owned_progress']) <= 64, 'finite progress samples')
                        print(f'OWNED PROGRESS {args.engine} {args.phase} elapsed={elapsed:.1f}s stderr_bytes={size}', flush=True)
                        save()
                        next_progress = time.monotonic() + 30
                    require(not DUMP.exists(), 'new devcoredump stops engine')
                io.poll = poll
                try:
                    require(time.monotonic() - admitted_at < 30, 'fresh in-controller health launch')
                    startup_deadline = entry['spawn_begin_monotonic'] + 300
                    while True:
                        remaining = startup_deadline - time.monotonic()
                        require(remaining > 0, 'absolute300s startup deadline')
                        line = io.line(min(120, remaining))
                        r['startup'].append(line)
                        require(len(r['startup']) <= 1000 and not line.startswith('ERR'), 'bounded non-error startup')
                        if line.startswith('READY '):
                            r['ready_seen'] = True
                            r['ready_monotonic'] = time.monotonic()
                            with stderr.open('rb') as stream:
                                ready_bytes = stream.read((64 << 20) + 1)
                            require(len(ready_bytes) <= 64 << 20, 'READY stderr cap')
                            r['ready_stderr_observed_bytes'] = len(ready_bytes)
                            # A partial line remains on the strict side so no enum can straddle and disappear.
                            r['ready_stderr_offset'] = ready_bytes.rfind(b'\n') + 1
                            monitor = DiagnosticReader(r['ready_stderr_offset'])
                            break
                    r['startup_wall_seconds_observed'] = r['ready_monotonic'] - entry['spawn_begin_monotonic']
                    expected_info = dict(e['expected_info'])
                    if args.phase != 'startup':
                        expected_info['expert_slots'] = prior[0]['effective_info']['expert_slots']
                    r['effective_info'] = worker_info(r['startup'], args.engine, expected_info, args.workers)
                    require(Path('/proc', str(entry['pid']), 'exe').resolve() == Path(argv[0]).resolve() and
                            sha(Path('/proc', str(entry['pid']), 'exe')) == BINARIES[args.engine]['sha256'], 'owned exact model executable')
                    require(Path('/proc', str(entry['pid']), 'cmdline').read_bytes().split(b'\0')[:-1] == [x.encode() for x in argv], 'actual argv')
                    actual_env = dict(x.decode().split('=', 1) for x in Path('/proc', str(entry['pid']), 'environ').read_bytes().split(b'\0') if b'=' in x)
                    require(actual_env == env, 'actual complete explicit environment; no inherited secrets/flags')
                    with stderr.open('rb') as stream:
                        initial_stderr = stream.read(r['ready_stderr_offset']).decode('utf-8', errors='replace')
                    r['selected_device_stderr_lines'] = [line for line in initial_stderr.splitlines()
                        if any(re.fullmatch(pattern, line) for pattern in e['expected_device_stderr_patterns'])]
                    require(all(any(re.fullmatch(pattern, line) for line in initial_stderr.splitlines())
                                for pattern in e['expected_device_stderr_patterns']),
                            'actual source-format selected device/backend identity')
                    task_line = f'strata generate: CPU pool tasks/phase: 18 (capped by rows), participating threads: {args.workers + 1}'
                    require(initial_stderr.splitlines().count(task_line) == 1, 'actual fixed18 tasks and selected participants')
                    r['pool_task_line'] = task_line
                    tids = sorted(Path('/proc', str(entry['pid']), 'task').iterdir())
                    require(0 < len(tids) <= 128, 'bounded actual owned model thread census')
                    r['ready_thread_affinity'] = {v.name: sorted(os.sched_getaffinity(int(v.name))) for v in tids}
                    r['startup_diagnostics'] = diagnostics(initial_stderr, startup=True)
                    r['startup_protocol_diagnostics'] = diagnostics('\n'.join(r['startup']), startup=True)
                    require(not r['startup_diagnostics']['severe_messages'] and
                            not r['startup_protocol_diagnostics']['severe_messages'], 'selected-path fatal/runtime gate')
                    capture('ready', entry, io.poll)
                    source_guard('ready')
                    r['post_ready_diagnostics'] = post_ready_diagnostics()
                    require(not r['post_ready_diagnostics']['severe_messages'], 'no new post-READY runtime errors')
                    if args.phase != 'startup':
                        require(entry['spawn_begin_monotonic'] + entry['wall_seconds'] - time.monotonic() > 90,
                                'actual owned deadline has graceful STOP reserve before GEN')
                        r['gen_send_attempted'] = True
                        r['gen_send_monotonic'] = time.monotonic()
                        save()
                        io.send(('GEN 16 ckpt=0 logprobs=5 ' + ','.join(map(str, tokens)) + '\n').encode())
                        r['gen_sent_monotonic'] = time.monotonic()
                        while True:
                            remaining = entry['spawn_begin_monotonic'] + entry['wall_seconds'] - time.monotonic()
                            require(remaining > 0, 'absolute model deadline')
                            line = io.line(remaining)
                            at = time.monotonic()
                            r['request_protocol'].append(line)
                            r['protocol_events'].append(dict(line=line, monotonic=at))
                            require(len(r['request_protocol']) <= 1000, 'bounded GEN protocol')
                            if line.startswith('ERR'):
                                r['request_error_seen'] = True
                                raise ValueError('selected-path protocol ERR: ' + line)
                            if line.startswith('PP '):
                                print(line, flush=True)
                                save()
                            if line.startswith('REUSED '):
                                r['prefill_end_monotonic'] = at
                            if line.startswith('T ') and 'first_token_monotonic' not in r:
                                r['first_token_monotonic'] = at
                            if line.startswith('DONE '):
                                r['request_done_seen'] = True
                                r['done_monotonic'] = at
                                break
                        require(not r.get('budget_stop_requested'), 'budget cancellation remains FAILED/incomplete')
                        r['post_ready_diagnostics'] = post_ready_diagnostics()
                        require(not r['post_ready_diagnostics']['severe_messages'], 'no selected-window runtime errors')
                        r['validation'] = validate(r['request_protocol'], args.engine, stderr.read_text(),
                                                    require_trace=not (args.phase.startswith('timed') or args.phase.startswith('heldout')),
                                                    expected_prompt_tokens=r['input']['tokens'])
                        r['host_wall_samples'] = dict(
                            prefill_to_REUSED_seconds=r['prefill_end_monotonic'] - r['gen_send_monotonic'],
                            decode_REUSED_to_DONE_seconds=r['done_monotonic'] - r['prefill_end_monotonic'],
                            ttft_seconds=r['first_token_monotonic'] - r['gen_send_monotonic'],
                            full_GEN_seconds=r['done_monotonic'] - r['gen_send_monotonic'],
                            scope='Observed protocol wall boundaries, includes pipe/scheduling/tracing overhead; engine DONE timings retained separately')
                        capture('done', entry, io.poll)
                        source_guard('done')
                    normal_quit(io, r, 'startup READY only' if args.phase == 'startup' else 'completed GEN16')
                    save()
                except BaseException as error:
                    r['primary_model_failure'] = repr(error)
                    save()
                    # READY idle, DONE, or an ERR response permits a bounded normal QUIT attempt.
                    if r['ready_seen'] and (not r['gen_send_attempted'] or r['request_done_seen'] or r.get('request_error_seen')):
                        io.poll = owner_poll
                        try:
                            normal_quit(io, r, 'preserved failure at responsive boundary')
                            save()
                            return
                        except BaseException as close_error:
                            r['gate_close_error'] = repr(close_error)
                            save()
                    raise
            r['model_launch_attempted'] = True
            save()
            so, se = run('engine', argv, selected_env=env, interaction=body,
                         wall=r['model_wall_seconds'], rss=88 << 30)
            r['decode_timing_lines'] = [line for line in se.splitlines() if line.startswith('strata decode timing: ')]
            require(not r['decode_timing_lines'], 'quiet sampling excludes decode tracing')
            r['decode_gpu_stage_lines'] = [line for line in se.splitlines() if line.startswith('strata decode GPU stages')]
            r['pool_overlap_summary_lines'] = [line for line in se.splitlines() if re.search(r'pool multi|R4 overlap|CPU pool call|R4 host', line)]
            r['service_event_lines'] = [line for line in se.splitlines() if line.startswith('strata prefill service ')]
            r['native_group_lines'] = [line for line in se.splitlines() if line.startswith('[prefill-native-dequant-group-v2]')]
            r['transfer_lines'] = [line for line in se.splitlines() if line.startswith('strata prefill expert H2D:')]
            require(not r['native_group_lines'] and not r['transfer_lines'], 'quiet calibration excludes transfer/group tracing')
            if args.profile_arm == '1':
                require(r['service_event_lines'], 'actual production service diagnostic output required')
                summary_lines = [line for line in r['service_event_lines'] if line.startswith('strata prefill service events: ')]
                category_lines = [line for line in r['service_event_lines'] if line.startswith('strata prefill service category: ')]
                require(len(summary_lines) == 1 and category_lines, 'one actual nonempty Prefill scope')
                summary = dict(word.split('=',1) for word in summary_lines[0].split(': ',1)[1].split())
                categories = [dict(word.split('=',1) for word in line.split(': ',1)[1].split()) for line in category_lines]
                require(summary['status'] == 'complete' and summary['checked_boundary'] == '1' and
                        all(summary[key] == '0' for key in ['overflow','pending','invalid_timestamp','query_errors','excluded_graph','bookkeeping_failed']),
                        'actual selected leaf event timing complete and positive')
                require(sum(int(row['count']) for row in categories) == sum(int(row['valid']) for row in categories) == int(summary['captured']),
                        'all actual captured events reconcile to valid category counts')
                require({'memcpy','grouped_gemm','iq_dequant_gu','iq_dequant_down'} <= {row['kind'] for row in categories},
                        'actual production copies, grouped GEMM and native dequant categories present')
                r['service_summary'] = summary
                r['service_categories'] = categories
            else:
                require(not r['service_event_lines'], 'profiling-off has no service collection')

            if args.phase != 'startup':
                require(r['validation']['token_logprob_sha256'] == native_proof['validation']['token_logprob_sha256'],
                        'all16 output IDs/LP5 exactly equal admitted baseline')
            r['final_diagnostics'] = post_ready_diagnostics()
            r['protocol_diagnostics'] = diagnostics('\n'.join(r['request_protocol']))
            require(r['model_normal_close'] and not r.get('primary_model_failure') and
                    not r['final_diagnostics']['severe_messages'] and not r['protocol_diagnostics']['severe_messages'], 'normal QUIT and selected path success')
            kernel('kernel-after-engine')
            health('health-after')
            kernel('kernel-after-posthealth')
            source_guard('final')
            require(all(sha(p) == digest for p, digest in r['prior_receipt_pins'].items()), 'immutable prior runs')
            if index > 0 and index % 2 == 0:
                r['pair_comparison'] = compare_pair(prior[-1], r,
                    str(roots[index - 1] / 'record.json'), sha(roots[index - 1] / 'record.json'))
                require(r['pair_comparison']['token_logprob_equal'], 'guard neutral token/LP5 fingerprint')
            r['complete'] = True
        except BaseException as error:
            r['error'] = repr(error)
        finally:
            if cursor and owner.active is None and guard.active is None:
                try:
                    kernel('kernel-final')
                    r['kernel_gate_passed'] = True
                except BaseException as error:
                    r['kernel_gate_error'] = repr(error)
            r['active'] = owner.active is not None or guard.active is not None
            r['boot_unchanged'] = Path('/proc/sys/kernel/random/boot_id').read_text().strip() == boot
            r['passed'] = bool(r['complete'] and not r['active'] and not r.get('error') and
                not r.get('primary_model_failure') and r.get('kernel_gate_passed') and r['boot_unchanged'] and
                all(strong(c) and c['exit_code'] == 0 for c in r['commands'] + r['guard_commands']))
            r['finished_utc'] = owner_module.utc()
            r['performance_sample_eligible'] = bool(r['passed'] and args.profile_arm == '0')
            r['calibration_sample_eligible'] = bool(r['passed'])
            r['performance_eligible'] = r['performance_sample_eligible']
            r['screening_only'] = True
            save()
        if args.phase == 'heldout-on' and r['passed']:
            samples = [closed(load(root / 'record.json')) for root in roots]
            pairs = [value for value in samples if value.get('pair_comparison')]
            require(len(pairs) == 5 and all(value['pair_comparison']['token_logprob_equal'] for value in pairs),
                    'five completed pairs: debug +3 timed +held-out; either second arm')
            summary = dict(passed=True, complete=True, active=False, main_reviewed=False,
                source_only_implementation=False, model_runs_executed=True,
                qualification_sha256=args.qualification_sha256, boot_id=boot,
                individual_receipts={str(root / 'record.json'): sha(root / 'record.json') for root in roots},
                individual_samples=[dict(phase=value['phase'], passed=value['passed'],
                    input=value.get('input'), validation=value.get('validation'),
                    host_wall_samples=value.get('host_wall_samples'),
                    performance_sample_eligible=value.get('performance_sample_eligible', False),
                    performance_eligible=value.get('performance_eligible', False),
                    pair_comparison=value.get('pair_comparison')) for value in samples],
                prompt_geometry=dict(debug_timed=32768, held_out=samples[-1]['input']['tokens']),
                held_out_provenance=q['held_out_prompt']['preparation_receipt'],
                scope='Neutral reported16 tokens/LP5; debug/timed32K and complete held-out32768..65536IDs. Three timed pairs only; no full-logit/quality/256K/optimal speed claim')
            require(not (NAMESPACE / 'campaign-summary.json').exists(), 'new compact summary only')
            compact = json.dumps(summary, indent=2) + '\n'
            require(len(compact.encode()) <= 1 << 20, 'compact campaign summary1MiB cap')
            (NAMESPACE / 'campaign-summary.json').write_text(compact)
        print(json.dumps({k: r.get(k) for k in ('passed', 'error', 'engine', 'phase', 'model_launch_attempted')}))
        return 0 if r['passed'] else 1
