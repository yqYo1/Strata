import argparse
import hashlib
import importlib.machinery
import importlib.util

from pathlib import Path
import sys

from unittest import mock

sys.dont_write_bytecode = True
SOURCE = Path(__file__).with_name('worker-long-decode-v1.py.txt')
loader = importlib.machinery.SourceFileLoader('long_decode_source_under_test', str(SOURCE))
spec = importlib.util.spec_from_loader(loader.name, loader)
module = importlib.util.module_from_spec(spec)
loader.exec_module(module)


def protocol():
    rows = ['RESUME 0'] + [f'PP {end} 32768 1 0' for end in (8192, 16384, 24576, 32767)] + ['REUSED 0']
    for i in range(128):
        rows += [f'T {100 + i}', f'LP -0.1 {100 + i}:-0.1 1000:-1 1001:-2 1002:-3 1003:-4']
    return rows + ['DONE 128 32768 100 200 length 0 0 0 2 3 4 0 0.0 32768 0']


def admission():
    return dict(long_decode128=dict(main_reviewed=True, status='passed', generated_tokens=128,
        prompt_tokens=32768, pool_tasks=18, first16_sha256=module.FIRST16_SHA,
        namespace=str(module.CLEAN_PARENT), order=[dict(workers=3, gu_arm='unchanged')]))


class LongDecodeFixtures(unittest.TestCase):
    def parse(self, rows):
        return module.validate_long(rows, 'xe', '', require_trace=False)

    def test_full128_and_actual_prefix_fingerprints(self):
        rows = protocol()
        value = self.parse(rows)
        self.assertEqual(len(value['output_ids']), 128)
        self.assertEqual(value['generated_tokens'], 128)
        full = '\n'.join(rows[6:-1]) + '\n'
        prefix = '\n'.join(rows[6:38]) + '\n'
        self.assertEqual(value['token_logprob_sha256'], hashlib.sha256(full.encode()).hexdigest())
        self.assertEqual(value['first16_token_logprob_sha256'], hashlib.sha256(prefix.encode()).hexdigest())
        self.assertNotEqual(value['first16_token_logprob_sha256'], module.FIRST16_SHA)
        self.assertIsNone(value['actual_route'])
        self.assertTrue(value['inferred_from_source'])

    def test_full_malformed_partial_duplicates_and_end_status(self):
        original = protocol()
        cases = []
        cases += [original[:-1], original[:38] + [original[-1]], original + ['DONE 128 32768 1 1 length']]
        for field in ('DONE 16 32768 100 200 length 0 0 0 2 3 4 0 0.0 32768 0',
                      'DONE 128 32768 100 200 cancel 0 0 0 2 3 4 0 0.0 32768 0',
                      'DONE 128 32768 100 200 stop 0 0 0 2 3 4 0 0.0 32768 0',
                      'DONE 128 32768 100 200 length 0 0 0 2 3 4 0 0.0 32767 0',
                      'DONE 128 32768 100 200 nonfinite 0 0 0 2 3 4 0 0.0 32768 0',
                      'DONE 128 32768 nan 200 length 0 0 0 2 3 4 0 0.0 32768 0'):
            cases.append(original[:-1] + [field])
        for token in ('T -1', 'T 248320', 'T 248044', 'T 248046'):
            case = list(original); case[6] = token; cases.append(case)
        for lp in ('LP nan 100:-1 1000:-2 1001:-3 1002:-4 1003:-5',
                   'LP -1 100:-1 100:-2 1001:-3 1002:-4 1003:-5',
                   'LP -1 248320:-1 1000:-2 1001:-3 1002:-4 1003:-5',
                   'LP -1 100:inf 1000:-2 1001:-3 1002:-4 1003:-5',
                   'LP -1 100:-1 1000:-2'):
            case = list(original); case[7] = lp; cases.append(case)
        case = list(original); case.insert(8, original[7]); cases.append(case)  # duplicated LP command
        case = list(original); case[8] = original[7]; cases.append(case)
        case = list(original); case[6], case[7] = case[7], case[6]; cases.append(case)
        case = list(original); case[4] = 'PP 32766 32768 1 0'; cases.append(case)
        case = list(original); case[5] = 'REUSED 1'; cases.append(case)
        cases += [['ERR selected path'], ['BDONE 0 nonfinite']]
        for rows in cases:
            with self.subTest(rows=rows[-1:] if rows else []), self.assertRaises((ValueError, IndexError)):
                self.parse(rows)

    def test_new_order_and_pending_fail_before_identity_or_owner(self):
        self.assertEqual(module.long_plan(admission(), 1, 3, 'unchanged')[0]['workers'], 3)
        bad = admission(); bad['long_decode128']['status'] = 'PENDING'
        wrong = admission(); wrong['long_decode128']['generated_tokens'] = 16
        paired = admission(); paired['long_decode128']['order'][0]['gu_arm'] = 'on'
        for q, sample, worker, arm in (({}, 1, 3, 'unchanged'), (bad, 1, 3, 'unchanged'), (wrong, 1, 3, 'unchanged'),
                                     (admission(), 1, 5, 'unchanged'), (admission(), 2, 3, 'unchanged'),
                                     (paired, 1, 3, 'on')):
            with mock.patch.object(module, 'identity', side_effect=AssertionError('file/process admission reached')):
                with self.assertRaises((ValueError, KeyError)):
                    module.long_admission(q, sample, worker, arm)


def owned_mocks(root):
    module.require(root.is_absolute() and root.resolve().is_relative_to(module.B.resolve()) and not root.exists(),
                   'fresh private CPU fixture root only')
    module.require(module.sha(module.OWNER) == module.OWNER_SHA, 'owner pinned before fixture import')
    spec = importlib.util.spec_from_file_location('qualified_long_cpu_owner', module.OWNER)
    owner_module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(owner_module)
    root.mkdir(parents=True)
    env = dict(PATH='/usr/bin:/bin', HOME='/home/yayoi', LANG='C.UTF-8', LC_ALL='C.UTF-8')
    receipts = []
    for stop in (False, True):
        output = root / ('stop' if stop else 'normal')
        output.mkdir()
        owner = owner_module.Owner(output / 'commands')
        script = "import sys\nprint('READY 262144 stop', flush=True)\n" + \
                 "for line in sys.stdin:\n" + \
                 " if line.startswith('GEN 128 '):\n" + \
                 ("  continue\n" if stop else "  print(" + repr('\n'.join(protocol())) + ", flush=True)\n") + \
                 " if line.startswith('STOP'): print('DONE 0 32768 1 1 cancel', flush=True)\n" + \
                 " if line.startswith('QUIT'): break\n"
        record = dict(ready_seen=False, gen_sent_monotonic=None, request_done_seen=False, complete=False, passed=False)
        def body(io, entry):
            module.require(io.line(5) == 'READY 262144 stop', 'actual mock READY')
            record['ready_seen'] = True
            io.send(b'GEN 128 ckpt=0 logprobs=5 1,2,3\n')
            record['gen_sent_monotonic'] = 0.0
            if stop:
                module.request_budget_stop(io, entry, record, lambda: None,
                    entry['spawn_begin_monotonic'] + entry['wall_seconds'] - 89)
                module.require(io.line(5).startswith('DONE 0 '), 'actual cancellation boundary')
                module.require(not module.request_budget_stop(io, entry, record, lambda: None,
                    entry['spawn_begin_monotonic'] + entry['wall_seconds'] - 88), 'STOP once only')
            else:
                rows = []
                while not rows or not rows[-1].startswith('DONE '):
                    rows.append(io.line(5))
                record['validation'] = module.validate_long(rows, 'xe', '', require_trace=False)
            record['request_done_seen'] = True
            module.normal_quit(io, record, 'CPU mock only, cancellation not model PASS')
        command, _, _ = owner.run('mock', [sys.executable, '-B', '-c', script], env, module.HERE,
                                  interaction=body, wall=30, cpu=5, text_cap=2 << 20,
                                  file_cap=2 << 20, total_cap=4 << 20, rss_cap=128 << 20)
        module.require(module.strong(command) and command['exit_code'] == 0, 'actual anchored mock QUIT/reap')
        if stop:
            module.require(record['budget_stop_requested'] and not record['complete'] and not record['passed'],
                           'cancellation kept incomplete/failed')
        receipts.append(dict(stop=stop, command=command, record=record))
    (root / 'cpu-owned-result.json').write_text(json.dumps(dict(gpu_executed=False, model_executed=False,
        controller_sha256=module.sha(SOURCE), fixture_sha256=module.sha(Path(__file__)),
        generated_tokens=128, commands=[row['command'] for row in receipts], owned_samples=receipts,
        active=False, complete=True, passed=True, main_reviewed=False), indent=2) + '\n')


if __name__ == '__main__':

    result = unittest.TextTestRunner(verbosity=2).run(unittest.defaultTestLoader.loadTestsFromTestCase(LongDecodeFixtures))
    if not result.wasSuccessful():

    if args.owned_root:
        owned_mocks(args.owned_root)
