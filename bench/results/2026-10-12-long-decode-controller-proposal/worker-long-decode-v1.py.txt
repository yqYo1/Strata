def validate_long(lines, engine, stderr, *, require_trace=True, expected_prompt_tokens=32768, expected_generated_tokens=128):
    require(engine == 'xe' and expected_prompt_tokens == 32768 and expected_generated_tokens == 128,
            'only actual32K fullGEN128 admitted')
    m = expected_generated_tokens
    n = expected_prompt_tokens
    require(type(n) is int and 32768 <= n <= 65536, 'bounded actual prompt geometry32768..65536')
    require(all(type(line) is str and len(line.encode()) <= 4096 for line in lines), 'bounded compact protocol fields')
    require(lines and all(type(x) is str and not x.startswith(('ERR', 'BDONE', 'B ')) for x in lines), 'no selected-path failure/batch protocol')
    allowed = ('RESUME ', 'PP ', 'REUSED ', 'T ', 'LP ', 'DONE ')
    require(all(x.startswith(allowed) for x in lines), 'only source-qualified GEN tags')
    require([x for x in lines if x.startswith('RESUME ')] == ['RESUME 0'] and
            [x for x in lines if x.startswith('REUSED ')] == ['REUSED 0'], 'fresh token0 request with no reuse')
    done = [x.split() for x in lines if x.startswith('DONE ')]
    require(len(done) == 1 and len(done[0]) == 16 and lines[-1].startswith('DONE '), 'one complete final DONE')
    d = done[0]
    require(int(d[1]) == m and int(d[2]) == n and int(d[14]) == n and int(d[8]) == 0 and
            d[5] == 'length' and list(map(int, d[6:8])) == [0, 0], 'full fresh GEN128, no draft/EOS truncation')
    require(all(math.isfinite(float(d[i])) and float(d[i]) > 0 for i in (3, 4)), 'finite positive engine samples')
    require(all(int(d[i]) >= 0 for i in (9, 10, 11, 12, 15)) and math.isfinite(float(d[13])) and float(d[13]) >= 0,
            'finite nonnegative tier accounting')
    pp = [x.split() for x in lines if x.startswith('PP ')]
    ends = list(range(8192, n - 1, 8192)) + [n - 1] if engine == 'xe' else [n - 1]
    require(all(len(x) == 5 for x in pp), 'complete PP fields')
    require([int(x[1]) for x in pp] == ends and all(len(x) == 5 and int(x[2]) == n and
            all(math.isfinite(float(v)) and float(v) >= 0 for v in x[3:]) for x in pp), 'actual full prompt route progress')
    require(len(lines) == 1 + len(ends) + 1 + 2 * m + 1 and lines[0] == 'RESUME 0' and
            lines[1 + len(ends)] == 'REUSED 0' and all(x.startswith('PP ') for x in lines[1:1 + len(ends)]),
            'strict fresh prompt/output phase ordering')
    output = [x for x in lines if x.startswith(('T ', 'LP '))]
    require(len(output) == 2 * m, 'complete128 token/LP5 pairs')
    ids = []
    for i in range(m):
        token, lp = output[2 * i].split(), output[2 * i + 1].split()
        require(len(token) == 2 and token[0] == 'T' and 0 <= int(token[1]) < 248320, 'actual token domain/pair')
        require(len(lp) == 7 and lp[0] == 'LP' and math.isfinite(float(lp[1])), 'finite LP5')
        top = []
        for field in lp[2:]:
            tid, value = field.split(':')
            require(0 <= int(tid) < 248320 and math.isfinite(float(value)), 'LP5 vocabulary/value')
            top.append(int(tid))
        require(len(set(top)) == 5, 'distinct top5 IDs')
        require(int(token[1]) not in EOS_IDS, 'no emittedEOS token')
        ids.append(int(token[1]))
    chunks = re.findall(r'^strata trace: prompt chunk (\d+) of (\d+)$', stderr, re.M)
    reads = re.findall(r'^strata trace: read (\d+) tokens \((windows|batched)\) in ([0-9.]+) ms$', stderr, re.M)
    if not require_trace:
        require(not chunks and not reads, 'clean timed route has tracing disabled')
    elif engine == 'xe':
        require(chunks == [(str(start), str(n - 1)) for start in range(0, n - 1, 8192)] and
                len(reads) == 1 and reads[0][:2] == (str(n - 1), 'batched'), 'actual Xe all8192chunks through prompt n-1')
    else:
        require(not chunks and len(reads) == 1 and reads[0][:2] == (str(n - 1), 'windows'), 'official only serial read_windows, no batched route')
    require(all(math.isfinite(float(x[2])) and float(x[2]) >= 0 for x in reads), 'finite individual traced part sample')
    encoded = ('\n'.join(output) + '\n').encode()
    return dict(prompt_tokens=n, source_bound_prefill_chunks=(n - 2) // 8192 + 1 if engine == 'xe' else None,
                observed_prefill_chunks=len(chunks) if require_trace and engine == 'xe' else None, generated_tokens=m, finish='length', output_ids=ids,
                token_logprob_sha256=hashlib.sha256(encoded).hexdigest(), token_logprob_lines=output,
                first16_token_logprob_sha256=hashlib.sha256(('\n'.join(output[:32]) + '\n').encode()).hexdigest(),
                engine_prefill_ms=float(d[3]), engine_decode_ms=float(d[4]), pp_samples=pp,
                individual_read_part_samples=[dict(tokens=int(a), route=b, ms=float(c)) for a, b, c in reads],
                actual_route=('batched8192' if engine == 'xe' else 'serial_read_windows8') if require_trace else None,
                source_bound_route='batched8192' if engine == 'xe' else 'serial_read_windows8',
                inferred_from_source=not require_trace,
                route_evidence_scope='observed debug trace' if require_trace else 'source-bound only; clean tracing absent',
                scope='Exact128 reported tokens/LP5 only; separate actual first16 fingerprint; no full-logit, quality, RESTORE or optimal-speed claim')


def resolved_long(value):
    if isinstance(value, str):
        require(not any(tag in value.upper() for tag in ('PENDING', 'UNRESOLVED', 'TODO')), 'unresolved long admission')
    elif isinstance(value, dict):
        for item in value.values():
            resolved_long(item)
    elif isinstance(value, list):
        for item in value:
            resolved_long(item)

def worker_info(startup, engine, extra, workers):
    require(startup[-1].split() == ['READY', '262144', 'stop'], 'exact physical-context READY')
    lines = [s for s in startup if s.startswith('INFO ')]
    require(len(lines) == 1, 'one actual INFO')
    result = {}
    for word in lines[0].split()[1:]:
        require('=' in word, 'INFO key/value field')
        key, value = word.split('=', 1)
        require(key not in result, 'duplicate INFO field')
        result[key] = value
    slots = result.get('expert_slots', '')
    require(re.fullmatch(r'[0-9]+', slots) and 0 < int(slots) <= 48 * 512,
            'actual positive integer expert slots within model population')
    require(workers in (3,4,5), 'bounded worker sweep')
    extra = dict(extra)
    require(extra.get('pool_workers', '5') == '5', 'admitted baseline extra workers5')
    extra['pool_workers'] = str(workers)
    expected = dict(context='262144', kv='int8', kv_resident='32768', spec='4' if engine == 'xe' else '8',
                    lookup='0', pool_workers=str(workers), pcie_frac='0.00', conversation_cache_mib='0')
    require(all(k not in expected or str(v) == expected[k] for k, v in extra.items()), 'Main INFO cannot override fixed policy')
    expected.update(extra)
    require(all(result.get(k) == str(v) for k, v in expected.items()), 'actual effective fixed settings')
    require(int(result['vram_free_mib']) >= 512, 'READY free VRAM >=512MiB')
    require(result.get('batch_slots', '0') == '0', 'ordinary serial request only')
    return result

                reference = closed(load(reference_path))
                require(reference['input']['sha256'] == r['input']['sha256'] and reference['binary'] == r['binary'],
                        'full128 comparison same input and qualified binary')
                equal = reference['validation']['token_logprob_sha256'] == r['validation']['token_logprob_sha256']
                r['long_output_reference'] = dict(path=str(reference_path), sha256=sha(reference_path),
                    full128_token_logprob_equal=equal, scope='Reported128 output/LP5 only, not full vocabulary')
                require(equal, 'all workers/optionalGU arms preserve newly observed full128 output fingerprint')
            if args.sample > 1 and args.paired_gu != 'unchanged':
                preceding = plan[args.sample - 2]
                if preceding['workers'] == args.workers and {preceding['gu_arm'], args.paired_gu} == {'off', 'on'}:
                    previous_path = previous_roots[-1] / 'xe-timed1-on/record.json'
                    previous = closed(load(previous_path))
                    require(previous['input']['sha256'] == r['input']['sha256'] and
                            previous['input']['tokens'] == r['input']['tokens'] == 32768 and
                            previous['worker_arm'] == r['worker_arm'] and previous['pool_tasks'] == r['pool_tasks'] == 18 and
                            previous['binary'] == r['binary'] and
                            {k: v for k, v in previous['environment'].items() if k != 'STRATA_IQ256_PAIRED_GU'} ==
                            {k: v for k, v in r['environment'].items() if k != 'STRATA_IQ256_PAIRED_GU'},
                            'adjacent GU pair same input/worker/binary/safety environment')
                    equal = previous['validation']['token_logprob_sha256'] == r['validation']['token_logprob_sha256']
                    r['paired_gu_comparison'] = dict(previous_path=str(previous_path), previous_sha256=sha(previous_path),
                        order=[preceding['gu_arm'], args.paired_gu], full128_token_logprob_equal=equal,
                        scope='128 reported tokens/LP5 only; no full vocabulary or quality parity')
                    require(equal, 'optional pairedGU full128 token/LP5 equality')
            r['complete'] = True
        except BaseException as error:
            r['error'] = repr(error)
        finally:
            if cursor and owner.active is None and guard.active is None:
                try:
                    kernel('kernel-final')
                    r['kernel_gate_passed'] = True
                except BaseException as error:
                    r['kernel_gate_error'] = repr(error)
            r['active'] = owner.active is not None or guard.active is not None
            r['boot_unchanged'] = Path('/proc/sys/kernel/random/boot_id').read_text().strip() == boot
            r['passed'] = bool(r['complete'] and not r['active'] and not r.get('error') and
                not r.get('primary_model_failure') and r.get('kernel_gate_passed') and r['boot_unchanged'] and
                all(strong(c) and c['exit_code'] == 0 for c in r['commands'] + r['guard_commands']))
            r['finished_utc'] = owner_module.utc()
            r['performance_sample_eligible'] = bool(r['passed'] and args.profile_arm == '0')
            r['calibration_sample_eligible'] = bool(r['passed'])
            r['performance_eligible'] = r['performance_sample_eligible']
            r['screening_only'] = True
            r['scope'] = '128 reported tokens/LP5; actual first16 prefix match only. No full-vocab quality/64K/256K/optimal-speed claim.'
            save()
        print(json.dumps({k: r.get(k) for k in ('passed', 'error', 'engine', 'phase', 'model_launch_attempted')}))
        return 0 if r['passed'] else 1
