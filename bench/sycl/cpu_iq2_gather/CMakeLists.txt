cmake_minimum_required(VERSION 3.22)
project(strata_cpu_iq2_gather LANGUAGES C CXX)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../cpu_roofline" cpu EXCLUDE_FROM_ALL)
get_filename_component(STRATA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(cpu_iq2_gather main.cpp candidate.cpp)
target_compile_features(cpu_iq2_gather PRIVATE cxx_std_20)
target_include_directories(cpu_iq2_gather PRIVATE "${STRATA_ROOT}/third_party/ggml")
target_compile_options(cpu_iq2_gather PRIVATE -march=znver3 -mtune=znver3 -mavx2 -mfma -mf16c)
target_link_libraries(cpu_iq2_gather PRIVATE expert_cpu)
