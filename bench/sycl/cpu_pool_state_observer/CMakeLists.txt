cmake_minimum_required(VERSION 3.22)
project(strata_cpu_pool_state_observer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../cpu_roofline" cpu EXCLUDE_FROM_ALL)
get_filename_component(STRATA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
get_target_property(PRODUCTION_CPU expert_cpu SOURCES)
list(REMOVE_ITEM PRODUCTION_CPU "${STRATA_ROOT}/src/kernels/cpu/pool.cpp")
add_library(expert_cpu_observed STATIC ${PRODUCTION_CPU} pool.cpp)
target_include_directories(expert_cpu_observed PUBLIC "${STRATA_ROOT}/include" PRIVATE "${STRATA_ROOT}/third_party/ggml")
target_compile_definitions(expert_cpu_observed PUBLIC STRATA_NATIVE_EXPERTS=1)
target_compile_options(expert_cpu_observed PRIVATE -march=znver3 -mtune=znver3)
set(CPU "${STRATA_ROOT}/src/kernels/cpu")
set_source_files_properties(${CPU}/expert.cpp ${CPU}/iq_avx512.cpp PROPERTIES
  COMPILE_OPTIONS "-mavx512f;-mavx512bw;-mavx512vl;-mavx512dq;-mavx512vnni;-mavx512vbmi;-mf16c;-mfma")
set_source_files_properties(${CPU}/q2_avx2.cpp ${CPU}/iq_avx2.cpp ${CPU}/iq_single_avx2.cpp ${CPU}/iq2_pair_avx2.cpp ${CPU}/kq_avx2.cpp
  PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-mf16c")
target_link_libraries(expert_cpu_observed PUBLIC ggml-cpu ggml-base Threads::Threads)
foreach(arm IN ITEMS raw observed)
  add_executable(pool_state_${arm} main.cpp)
  target_include_directories(pool_state_${arm} PRIVATE "${STRATA_ROOT}/third_party/ggml")
  target_compile_options(pool_state_${arm} PRIVATE -march=znver3 -mtune=znver3 -mavx2 -mfma -mf16c)
endforeach()
target_compile_definitions(pool_state_raw PRIVATE POOL_OBSERVED=0)
target_compile_definitions(pool_state_observed PRIVATE POOL_OBSERVED=1)
target_link_libraries(pool_state_raw PRIVATE expert_cpu)
target_link_libraries(pool_state_observed PRIVATE expert_cpu_observed)
