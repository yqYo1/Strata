# Validation-only wrapper; original sycl/CMakeLists.txt is byte-identical to PR #6.
# New project files follow the repository LICENSE.
cmake_minimum_required(VERSION 3.24)
project(strata_pr6_validation LANGUAGES C CXX)
option(PR6_MAIN_REVIEWED_BUILD "Main has reviewed the historical route and owns execution" OFF)
if(NOT PR6_MAIN_REVIEWED_BUILD)
  message(FATAL_ERROR "Historical source is unsafe/unvalidated. Main must review before enabling this build.")
endif()
set(PR6_VENDOR_ROOT "" CACHE PATH "External historical DPCT + ggml-common header snapshot root")
if(NOT IS_ABSOLUTE "${PR6_VENDOR_ROOT}")
  message(FATAL_ERROR "PR6_VENDOR_ROOT must be an absolute external dependency snapshot; nothing is fetched here")
endif()
file(READ "${CMAKE_CURRENT_LIST_DIR}/../../bench/results/2026-10-16-pr6-faithful-restore/historical-closure.json" closure)
string(JSON vendor_count LENGTH "${closure}" external_vendor_files)
# Do not silently drop a requested translation unit from the historical closure.
string(JSON project_count LENGTH "${closure}" files)
if(NOT project_count EQUAL 198)
  message(FATAL_ERROR "Expected the complete 198-file historical closure")
endif()
math(EXPR project_last "${project_count} - 1")
foreach(i RANGE 0 ${project_last})
  string(JSON path GET "${closure}" files ${i} path)
  if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/${path}")
    message(FATAL_ERROR "Missing historical project source: ${path}")
  endif()
endforeach()
set(resolved_count 3) # explicit XMX, GGUF source and fused stub TUs
foreach(group _spec _cpu _cpu_native _core _kernels _engine _prefill _main)
  string(JSON group_count LENGTH "${closure}" resolver_groups ${group})
  math(EXPR resolved_count "${resolved_count} + ${group_count}")
  math(EXPR group_last "${group_count} - 1")
  foreach(i RANGE 0 ${group_last})
    string(JSON selected GET "${closure}" resolver_groups ${group} ${i} selected)
    if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/${selected}")
      message(FATAL_ERROR "Missing selected translation unit: ${selected}")
    endif()
  endforeach()
endforeach()
if(NOT resolved_count EQUAL 84)
  message(FATAL_ERROR "Expected all 84 historical translation units")
endif()
math(EXPR vendor_last "${vendor_count} - 1")
foreach(i RANGE 0 ${vendor_last})
  string(JSON path GET "${closure}" external_vendor_files ${i} path)
  string(JSON expected GET "${closure}" external_vendor_files ${i} sha256)
  if(NOT EXISTS "${PR6_VENDOR_ROOT}/${path}")
    message(FATAL_ERROR "Missing historical dependency: ${path}")
  endif()
  file(SHA256 "${PR6_VENDOR_ROOT}/${path}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Historical dependency hash mismatch: ${path}")
  endif()
endforeach()
set(STRATA_GGML_DIR "" CACHE PATH "External llama.cpp 3cf03257f219afbe7334045ff7c6a06ac68c627d checkout")
if(NOT IS_ABSOLUTE "${STRATA_GGML_DIR}" OR NOT EXISTS "${STRATA_GGML_DIR}/ggml/CMakeLists.txt")
  message(FATAL_ERROR "Supply the pinned offline llama.cpp checkout in STRATA_GGML_DIR; Main verifies HEAD and diff")
endif()
# Keep third-party dependencies outside the restored project tree. These paths
# supply the same historical include names; project header precedence is intact.
include_directories(SYSTEM "${PR6_VENDOR_ROOT}/sycl/include" "${PR6_VENDOR_ROOT}/third_party/ggml")
set(STRATA_SYCL_PARITY OFF CACHE BOOL "Historical parity suite is outside this engine-only closure")
if(STRATA_SYCL_PARITY)
  message(FATAL_ERROR "This closure restores target strata only; historical parity targets were not copied")
endif()
add_subdirectory(sycl)
